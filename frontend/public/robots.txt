# Un solo gruppo per tutti i crawler che possono leggere il sito, AI di ricerca compresi.
# Sono nominati uno per uno (oltre a "*") così si vede a colpo d'occhio che non sono bloccati,
# e condividono le stesse regole: un gruppo con un nome proprio ignora quelle di "*",
# quindi i Disallow qui sotto valgono per tutti.
#   OpenAI:      OAI-SearchBot (ricerca di ChatGPT), ChatGPT-User (pagina aperta da un utente)
#   Perplexity:  PerplexityBot (ricerca), Perplexity-User (pagina aperta da un utente)
#   Anthropic:   Claude-SearchBot (ricerca), Claude-User (pagina aperta da un utente)
User-agent: *
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: PerplexityBot
User-agent: Perplexity-User
User-agent: Claude-SearchBot
User-agent: Claude-User
Allow: /

# Block the admin dashboard (CMS login/content management)
Disallow: /dashboard/

# /api/ NON va bloccato: le pagine prerenderizzate, una volta caricate,
# richiedono di nuovo i dati a /api/v1 (articolo, progetti, flag). Googlebot
# rispetta robots.txt anche per queste richieste: con "Disallow: /api/" la
# chiamata falliva e la pagina diventava "Articolo non trovato", cioè 244
# articoli del blog segnalati come soft 404 (Search Console, 2026-11).
# Le risposte API non finiscono nell'indice comunque: Apache le serve con
# X-Robots-Tag: noindex (frontend/public/.htaccess).
Disallow: /auth/
Disallow: /analytics/
Disallow: /audit/
Disallow: /stats/
Disallow: /system/
Disallow: /cron/
